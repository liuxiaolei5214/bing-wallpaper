// ====================== Worker 主入口 ====================== export default { async fetch(request, env, ctx) { const url = new URL(request.url); const path = url.pathname; const method = request.method; // favicon 空响应 if (path === '/favicon.ico') { return new Response(null, { status: 204, headers: { 'Cache-Control': 'public, max-age=86400' } }); } // 触发青龙采集任务接口 if (path === '/api/trigger' && method === 'POST') { if (!validateApiKey(request)) { return Response.json({ success: false, error: "无效访问密钥" }, { status: 403 }); } const taskId = parseInt(url.searchParams.get('taskId')) || 10; const result = await triggerQLTask(taskId); return Response.json(result); } // 青龙提交壁纸数据接口 if (path === '/api/save' && method === 'POST') { if (!validateApiKey(request)) { return Response.json({ success: false, error: "无效访问密钥" }, { status: 403 }); } try { const data = await request.json(); const force = url.searchParams.get('force') === 'true'; const res = await saveBingData(env, data, force, true); return Response.json(res, { status: res.success ? 200 : 400 }); } catch (err) { return Response.json({ success: false, error: err.message }, { status: 500 }); } } // 删除指定日期壁纸数据 if (path === '/api/delete' && method === 'DELETE') { if (!validateApiKey(request)) { return Response.json({ success: false, error: "无效访问密钥" }, { status: 403 }); } const date = url.searchParams.get('date'); if (!date) { return Response.json({ success: false, error: "缺失date参数" }, { status: 400 }); } try { const year = new Date(date).getFullYear(); const yearKey = `${KV_KEYS.YEAR_PREFIX}${year}`; let list = await env.BING_DATA.get(yearKey, 'json'); if (!list || !Array.isArray(list)) list = []; list = list.filter(item => item.date !== date); await env.BING_DATA.put(yearKey, JSON.stringify(list)); // 同步清理今日缓存 const todayCache = await env.BING_DATA.get(KV_KEYS.TODAY, 'json'); if (todayCache && todayCache.date === date) { await env.BING_DATA.delete(KV_KEYS.TODAY); } return Response.json({ success: true, msg: "数据删除完成", targetDate: date }); } catch (err) { return Response.json({ success: false, error: err.message }, { status: 500 }); } } // 获取当日壁纸 if (path === '/api/today') { const res = await getTodayData(env); return Response.json(res); } // 根据日期单条查询 if (path === '/api/date') { const date = url.searchParams.get('date'); const res = await getDateData(env, date); return Response.json(res); } // 获取整年全部数据 if (path === '/api/year') { const year = url.searchParams.get('year'); const res = await getYearData(env, year); return Response.json(res); } // 按年月筛选数据 if (path === '/api/month') { const year = url.searchParams.get('year'); const month = url.searchParams.get('month'); if (!year || !month) { return Response.json({ success: false, error: "缺失year/month参数" }, { status: 400 }); } const raw = await getYearData(env, parseInt(year)); if (!raw.success) return Response.json(raw); const targetMonth = String(month).padStart(2, '0'); const filterList = raw.data.filter(item => { const d = new Date(item.date); const m = String(d.getMonth() + 1).padStart(2, '0'); return m === targetMonth; }); return Response.json({ success: true, year: parseInt(year), month: targetMonth, count: filterList.length, data: filterList }); } // 分页全量历史 if (path === '/api/history') { const limit = Number(url.searchParams.get('limit')) || 20; const page = Number(url.searchParams.get('page')) || 1; const year = url.searchParams.get('year') ? parseInt(url.searchParams.get('year')) : null; const res = await getHistory(env, limit, page, year); return Response.json(res); } // 获取所有存在数据的年份列表 if (path === '/api/years') { const res = await getAvailableYears(env); return Response.json(res); } // 首页渲染HTML页面 if (path === '/' || path === '') { const html = generateHTMLPage(); return new Response(html, { headers: { 'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'public, max-age=300' } }); } // 404兜底 return Response.json({ error: "接口不存在", path: path }, { status: 404 }); } };
